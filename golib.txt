        // Dark Mode Logic
        const themeToggleBtn = document.getElementById('theme-toggle');

        function toggleTheme() {
            darkIcon.classList.toggle('hidden');
            lightIcon.classList.toggle('hidden');
            if (localStorage.getItem('color-theme')) {
                if (localStorage.getItem('color-theme') === 'light') {
                    document.documentElement.classList.add('dark');
                    localStorage.setItem('color-theme', 'dark');
                } else {
                    document.documentElement.classList.remove('dark');
                    localStorage.setItem('color-theme', 'light');
                }
            } else {
                if (document.documentElement.classList.contains('dark')) {
                    document.documentElement.classList.remove('dark');
                    localStorage.setItem('color-theme', 'light');
                } else {
                    document.documentElement.classList.add('dark');
                    localStorage.setItem('color-theme', 'dark');
                }
            }
        }

                overlay.classList.add('hidden');
            }
        }
        openBtn.addEventListener('click', () => toggleSidebar(true));
        closeBtn.addEventListener('click', () => toggleSidebar(false));
        overlay.addEventListener('click', () => toggleSidebar(false));

        // Render Functions
        function renderRecommendations(filterCat = 'All Books', searchQuery = '') {
            const grid = document.getElementById('recGrid');
            grid.innerHTML = '';

            const filteredBooks = books.filter(book => {
                const matchCategory = filterCat === 'All Books' || book.category === filterCat;
                const matchSearch = book.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
                                    book.author.toLowerCase().includes(searchQuery.toLowerCase());
                return matchCategory && matchSearch;
            });

            if (filteredBooks.length === 0) {
                grid.innerHTML = `<p class="text-slate-500 py-10">Buku tidak ditemukan.</p>`;
                return;
            }

            filteredBooks.forEach(book => {
                const coverUrl = `https://placehold.co/400x600/${book.coverColor}/${book.textColor}?text=${encodeURIComponent(book.title.split(' ').join('\n'))}`;
                
                const card = document.createElement('div');
                card.className = "w-36 sm:w-48 flex-shrink-0 cursor-pointer snap-start group";
                card.onclick = () => showOverview(book);
                
                card.innerHTML = `
                    <div class="w-full aspect-[2/3] rounded-xl shadow-[0_15px_30px_-10px_rgba(0,0,0,0.2)] dark:shadow-[0_15px_30px_-10px_rgba(0,0,0,0.8)] overflow-hidden bg-slate-200 mb-3">
                        <img src="${coverUrl}" alt="${book.title}" class="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500">
                    </div>
                    <h4 class="font-bold text-slate-900 dark:text-white text-sm line-clamp-1">${book.title}</h4>
                    <p class="text-xs text-slate-500 dark:text-slate-400 mt-0.5">${book.author}</p>
                `;
                grid.appendChild(card);
            });
        }

        // Render Pill/Chip Category
        function renderCategories() {
            const grid = document.getElementById('catGrid');
            grid.innerHTML = '';
            
            const categories = ['All Books', ...new Set(books.map(b => b.category))];
            
            categories.forEach(cat => {
                const isActive = cat === activeCategory;
                const btn = document.createElement('button');
                
                btn.className = `px-5 py-2 rounded-full whitespace-nowrap text-sm font-semibold transition-all shadow-sm snap-start ${isActive ? 'bg-emerald-500 text-white shadow-[0_8px_15px_-5px_rgba(16,185,129,0.5)]' : 'bg-white dark:bg-slate-800/80 text-slate-600 dark:text-slate-300 border border-slate-100 dark:border-slate-700 hover:bg-slate-50 dark:hover:bg-slate-700'}`;
                btn.textContent = cat;
                
                btn.onclick = () => {
                    activeCategory = cat;
                    renderCategories(); // Update state aktif (warna tombol)
                    renderRecommendations(cat, document.getElementById('searchInput') ? document.getElementById('searchInput').value : '');
                };
                
                grid.appendChild(btn);
            });
        }

        // Search Handlers
        function executeSearch() {
            const query = document.getElementById('searchInput').value;
            renderRecommendations(activeCategory, query);
        }

        if(document.getElementById('searchInput')) {
            document.getElementById('searchInput').addEventListener('input', executeSearch);
        }
        if(document.getElementById('searchBtn')) {
            document.getElementById('searchBtn').addEventListener('click', executeSearch);
        }

        // Modal Logic
        function showOverview(book) {
            currentSelectedBook = book;
            const modal = document.getElementById('overviewModal');
            const coverUrl = `https://placehold.co/400x600/${book.coverColor}/${book.textColor}?text=${encodeURIComponent(book.title.split(' ').join('\n'))}`;
            
            document.getElementById('ovCover').src = coverUrl;
            document.getElementById('ovTitle').textContent = book.title;
            document.getElementById('ovAuthorName').textContent = book.author;
            document.getElementById('ovRating').textContent = book.rating;
            document.getElementById('ovPages').textContent = book.pages;
            document.getElementById('ovLang').textContent = book.language;
            document.getElementById('ovDesc').textContent = book.desc;

            modal.classList.remove('hidden');
        }

        document.getElementById('closeOverviewBtn').addEventListener('click', () => {
            document.getElementById('overviewModal').classList.add('hidden');
            currentSelectedBook = null;
        });

        document.getElementById('ovReadBtn').addEventListener('click', () => {
            document.getElementById('overviewModal').classList.add('hidden');
            if (currentSelectedBook) openBook(currentSelectedBook);
        });

        // Flipbook Logic
        function generateBookHTML(book) {
            let html = '';
            html += `
                <div class="page page-cover page-cover-top" data-density="hard" style="background-color: #${book.coverColor};">
                    <div class="page-content justify-center items-center text-white border-2 border-white/20 rounded-lg p-6 w-full h-full flex flex-col">
                        <h2 class="text-3xl font-bold mb-4">${book.title}</h2>
                        <p class="text-lg opacity-80">${book.author}</p>
                    </div>
                </div>
            `;

            for (let i = 1; i <= 6; i++) {
                html += `
                    <div class="page">
                        <div class="page-content">
                            <h3 class="text-xl font-bold text-slate-800 mb-4 border-b pb-2">Bab ${i}</h3>
                            <p class="text-slate-600 text-sm leading-relaxed mb-4 text-justify">
                                Ini adalah konten simulasi untuk halaman ke-${i}.
                            </p>
                            <div class="page-number">${i}</div>
                        </div>
                    </div>
                `;
            }

            html += `
                <div class="page page-cover page-cover-bottom" data-density="hard" style="background-color: #${book.coverColor};">
                    <div class="page-content justify-center items-center text-white/50">
                        <i class="fa-solid fa-qrcode text-6xl mb-4"></i>
                        <p>GoLib</p>
                    </div>
                </div>
            `;
            return html;
        }

        function openBook(book) {
            const modal = document.getElementById('flipbookModal');
            const renderer = document.getElementById('bookRenderer');
            document.getElementById('modalBookTitle').textContent = book.title;
            
            renderer.innerHTML = generateBookHTML(book);
            modal.classList.remove('hidden');

            if (pageFlipInstance) {
                pageFlipInstance.destroy();
                pageFlipInstance = null;
            }

            const isMobile = window.innerWidth < 768;
            
            setTimeout(() => {
                pageFlipInstance = new StPageFlip.PageFlip(renderer, {
                    width: isMobile ? 300 : 400,
                    height: isMobile ? 450 : 600,
                    size: "stretch",
                    minWidth: 250, maxWidth: 1000,
                    minHeight: 350, maxHeight: 1000,
                    maxShadowOpacity: 0.3,
                    showCover: true,
                    mobileScrollSupport: false,
                    usePortrait: true
                });
                pageFlipInstance.loadFromHTML(document.querySelectorAll('.page'));
            }, 100);
        }

        function closeModal() {
            document.getElementById('flipbookModal').classList.add('hidden');
            if (pageFlipInstance) {
                pageFlipInstance.destroy();
                pageFlipInstance = null;
            }
            document.getElementById('bookRenderer').innerHTML = '';
        }

        document.getElementById('closeModalBtn').addEventListener('click', closeModal);
        
        // Init
        window.onload = () => {
            renderRecommendations();
            renderCategories();
        };

    </script>
